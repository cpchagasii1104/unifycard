-- Tenants (multi-tenancy)
CREATE TABLE tenants (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  name TEXT NOT NULL,
  slug TEXT NOT NULL UNIQUE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now()

-- Actors (SSOT de identidade econômica)
CREATE TABLE actors (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  actor_type TEXT NOT NULL CHECK (actor_type IN ('person', 'company', 'system')),
  external_id TEXT,
  display_name TEXT NOT NULL,
  cpf_cnpj TEXT,
  kyc_status TEXT NOT NULL DEFAULT 'pending' CHECK (kyc_status IN ('pending', 'verified', 'rejected')),
  kyc_verified_at TIMESTAMPTZ,
  kyc_limit_cents BIGINT DEFAULT 500000, -- 5000 BRL
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  updated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE(tenant_id, external_id),
  -- HARD MODE: Constraint técnica para FK composta cross-tenant
  -- Necessária para garantir isolamento multi-tenant estrutural em referências
  CONSTRAINT actors_tenant_id_id_unique UNIQUE (tenant_id, id)
);

CREATE INDEX idx_actors_tenant ON actors(tenant_id);
CREATE INDEX idx_actors_cpf_cnpj ON actors(cpf_cnpj);
-- HARD MODE: Índice composto para performance da FK composta
CREATE INDEX idx_actors_tenant_id_id ON actors(tenant_id, id);

-- ATL (Anti-Terrorism/Laundering)
-- 
-- DECISÃO ARQUITETURAL: Bloqueio atual apenas (sem histórico)
--
-- atl_blocked_actors registra apenas o estado atual de bloqueio.
-- UNIQUE(actor_id) garante que um actor só pode estar bloqueado uma vez.
--
-- Se histórico de bloqueios for necessário no futuro:
--   - Remover UNIQUE(actor_id)
--   - Adicionar coluna blocked_until TIMESTAMPTZ
--   - Implementar lógica de expiração de bloqueio
--
-- Por ora, mantemos modelo simples: bloqueio atual apenas.
--
-- HARD MODE: FK composta cross-tenant para garantir isolamento
CREATE TABLE atl_blocked_actors (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  actor_id UUID NOT NULL, -- Removido UNIQUE e REFERENCES direto
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  blocked_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  blocked_reason TEXT NOT NULL,
  blocked_by UUID,
  -- HARD MODE: UNIQUE composta para permitir FK composta
  CONSTRAINT atl_blocked_actors_actor_unique UNIQUE (actor_id),
  -- HARD MODE: FK composta: proteção estrutural cross-tenant
  -- Garante que actor_id pertence ao mesmo tenant_id
  CONSTRAINT fk_atl_blocked_actors_actor
    FOREIGN KEY (tenant_id, actor_id)
    REFERENCES actors(tenant_id, id)
);

CREATE INDEX idx_atl_blocked_actor ON atl_blocked_actors(actor_id);
-- HARD MODE: Índice composto para performance da FK composta
CREATE INDEX idx_atl_blocked_actors_tenant_actor ON atl_blocked_actors(tenant_id, actor_id);

-- ============================================================
-- BANK LEDGER (Append-Only SSOT)
-- ============================================================
--
-- DECISÃO ARQUITETURAL: Ledger sem balance_after
--
-- bank_ledger é append-only puro:
--   - Não armazena balance_after calculado
--   - Saldo é calculado on-demand via SUM(direction, amount_cents)
--   - Não há constraint de saldo não-negativo no banco
--
-- Justificativa:
--   - Append-only é mais seguro contra drift
--   - Evita necessidade de UPDATE em cada INSERT
--   - Regras financeiras (ex: saldo mínimo) são aplicação, não banco
--   - Performance: INSERT simples, sem cálculo de saldo
--
-- Se blindagem hard de saldo for necessária no futuro:
--   - Adicionar balance_after BIGINT
--   - Adicionar CHECK (balance_after >= 0)
--   - Implementar trigger que calcula balance_after antes de INSERT
--
-- Por ora, mantemos append-only puro.
--

-- HARD MODE: FK composta cross-tenant completa (account_id + transaction_id)
--
-- DECISÃO ARQUITETURAL: transaction_id é opcional (pode ser NULL)
--
-- Ledger é append-only e pode registrar movimentações que não vêm de
-- bank_transactions (ex: ajustes manuais, correções, inicializações).
-- Se transaction_id for NULL, a FK composta não valida (comportamento padrão).
-- Isso é intencional: permite flexibilidade no registro de movimentações.
--
-- Se vinculação obrigatória a transaction for necessária:
--   - Alterar para: transaction_id UUID NOT NULL
--   - Garantir que todas as inserções tenham transaction_id válido
--
CREATE TABLE bank_ledger (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  account_id UUID NOT NULL, -- Removido REFERENCES direto
  transaction_id UUID, -- Removido REFERENCES direto (opcional por design)
  direction ledger_direction NOT NULL,  -- HARD MODE+: Padronizado como ENUM
  amount_cents BIGINT NOT NULL CHECK (amount_cents > 0),
  purpose transfer_purpose,
  justification TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  -- FK composta: proteção estrutural cross-tenant (account)
  CONSTRAINT fk_bank_ledger_account 
    FOREIGN KEY (tenant_id, account_id) 
    REFERENCES bank_accounts(tenant_id, id),
  -- HARD MODE: FK composta: proteção estrutural cross-tenant (transaction)
  -- Garante que transaction_id pertence ao mesmo tenant_id
  CONSTRAINT fk_bank_ledger_transaction
    FOREIGN KEY (tenant_id, transaction_id)
    REFERENCES bank_transactions(tenant_id, id)
);

CREATE INDEX idx_bank_ledger_account ON bank_ledger(account_id);

-- ============================================================
-- VIEW: System Coverage
-- ============================================================
--
-- HARD MODE: Join defensivo com tenant_id explícito
-- Mesmo que FK composta já garanta isolamento, reforçamos na view
-- para clareza estrutural e performance de query
--
-- NOTA DE ESCALA (Dia 2):
-- Esta view executa dois subqueries correlacionados independentes por INSERT,
-- cada um varrendo bank_accounts + bank_ledger para o tenant.
-- Em baixa escala: irrelevante. Em alta escala: candidato a
-- materialized view ou contador incremental particionado por tenant.
--

CREATE OR REPLACE VIEW system_coverage AS
SELECT 
  t.id as tenant_id,
  COALESCE((
    SELECT SUM(CASE WHEN bl.direction = 'credit' THEN bl.amount_cents ELSE -bl.amount_cents END)
    FROM bank_accounts ba
    JOIN bank_ledger bl 
      ON bl.account_id = ba.id
     AND bl.tenant_id = ba.tenant_id  -- HARD MODE: Join defensivo explícito
    WHERE ba.tenant_id = t.id AND ba.owner_type = 'system'
  ), 0) as execution_capacity_cents,
  COALESCE((
    SELECT SUM(CASE WHEN bl.direction = 'credit' THEN bl.amount_cents ELSE -bl.amount_cents END)
    FROM bank_accounts ba
    JOIN bank_ledger bl 
      ON bl.account_id = ba.id
     AND bl.tenant_id = ba.tenant_id  -- HARD MODE: Join defensivo explícito
    WHERE ba.tenant_id = t.id AND ba.owner_type != 'system'
  ), 0) as total_credits_cents
FROM tenants t;

-- ============================================================
-- TRIGGER: Coverage Check + Auditoria Real
-- ============================================================

CREATE OR REPLACE FUNCTION check_coverage_before_credit()
RETURNS TRIGGER
SET search_path = pg_catalog, pg_temp
AS $$
DECLARE
  v_coverage_bps INTEGER;
  v_owner_type TEXT;
  v_capacity BIGINT;
  v_credits BIGINT;
  v_system_account_id UUID;
  v_max_bps INTEGER;
BEGIN
  IF NEW.direction != 'credit' THEN RETURN NEW; END IF;
  
  -- HARD MODE: Filtro explícito por tenant_id para consistência arquitetural
  SELECT ba.owner_type INTO v_owner_type 
  FROM bank_accounts ba 
  WHERE ba.id = NEW.account_id
    AND ba.tenant_id = NEW.tenant_id;  -- HARD MODE: Filtro defensivo explícito

  IF v_owner_type = 'system' THEN RETURN NEW; END IF;
  
  -- HARD MODE++: Lock transacional no fundo de execução
  -- Serializa validação por tenant (evita race condition crítica)
  -- Escopo: apenas a conta 'system' do tenant, não o ledger inteiro
  --
  -- NOTA: ORDER BY created_at LIMIT 1 é redundante dado o partial unique index
  -- uq_bank_accounts_one_system_per_tenant (garante no máximo 1 row system/tenant).
  -- Mantido intencionalmente como defesa estrutural: se o índice for removido
  -- no futuro, o comportamento do trigger permanece correto e determinístico.
  -- Custo: zero. Risco de remover: silencioso.
  SELECT ba.id
  INTO v_system_account_id
  FROM bank_accounts ba
  WHERE ba.tenant_id = NEW.tenant_id
   AND ba.owner_type = 'system'
  ORDER BY ba.created_at
  LIMIT 1
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'SYSTEM_ACCOUNT_NOT_FOUND: %', NEW.tenant_id
    USING ERRCODE = 'P0001';
  END IF;
  
  -- HARD MODE++: COALESCE defensivo para edge case de tenant sem histórico
  SELECT COALESCE(execution_capacity_cents, 0),
         COALESCE(total_credits_cents, 0)
  INTO v_capacity, v_credits
  FROM system_coverage 
  WHERE tenant_id = NEW.tenant_id;
  
  -- HARD MODE++: Inclui o crédito que está sendo inserido (validação prospectiva)
  -- Valida o estado REAL após o commit, não o estado atual
  v_credits := v_credits + NEW.amount_cents;
  
  IF v_capacity > 0 THEN
    v_coverage_bps := CEIL((v_credits::numeric * 10000) / v_capacity);
  ELSE
    v_coverage_bps := 10000;
  END IF;
  
  -- HARD MODE++: Governança híbrida (default institucional + override por tenant)
  -- Usa regra da tabela execution_fund_rules se existir, senão usa default institucional
  --
  -- NOTA DE MECANISMO (blindagem constitucional):
  -- SELECT INTO sem row encontrada NÃO executa a expressão (* 100).
  -- A variável v_max_bps permanece no estado inicial (NULL).
  -- Como max_percentage_of_reserve é NOT NULL, o único caso real de NULL
  -- aqui é NOT FOUND (tenant sem regra configurada).
  -- O check explícito abaixo documenta essa intenção, não apenas o resultado.
  SELECT max_percentage_of_reserve * 100
  INTO v_max_bps
  FROM execution_fund_rules
  WHERE tenant_id = NEW.tenant_id;

  IF NOT FOUND OR v_max_bps IS NULL THEN
    v_max_bps := 8000; -- Default institucional (80%)
  END IF;
  
  IF v_coverage_bps >= v_max_bps THEN
    RAISE EXCEPTION 'COVERAGE_EXCEEDED: % bps cobertura (limite: % bps)', 
      v_coverage_bps, v_max_bps
      USING ERRCODE = 'P0001';
  END IF;
  
  -- AUDITORIA REAL: Registra verificação aprovada.
  -- Tentativas bloqueadas não chegam aqui: RAISE EXCEPTION acima aborta a TX
  -- e reverte qualquer INSERT anterior, incluindo este.
  INSERT INTO coverage_audit_log (
    tenant_id, coverage_bps, execution_capacity_cents, 
    total_credits_cents
  ) VALUES (
    NEW.tenant_id, v_coverage_bps, v_capacity, 
    v_credits
  );
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_coverage
  BEFORE INSERT ON bank_ledger
  FOR EACH ROW EXECUTE FUNCTION check_coverage_before_credit();

-- ============================================================
-- TRIGGER: Consistência Monetária de Splits
-- ============================================================
--
-- HARD MODE++: Blindagem contra inconsistência intra-transação
-- Garante que a soma dos splits não exceda o valor da transação
--

CREATE OR REPLACE FUNCTION validate_split_consistency()
RETURNS TRIGGER
SET search_path = pg_catalog, pg_temp
AS $$
DECLARE
  v_transaction_amount BIGINT;
  v_splits_sum BIGINT;
BEGIN
  -- Lock na transação para serializar validação
  SELECT amount_cents
  INTO v_transaction_amount
  FROM bank_transactions
  WHERE tenant_id = NEW.tenant_id
   AND id = NEW.transaction_id
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'TRANSACTION_NOT_FOUND: %', NEW.transaction_id
      USING ERRCODE = 'P0001';
  END IF;

  -- Soma todos os splits exceto o atual (em caso de UPDATE)
  SELECT COALESCE(SUM(amount_cents), 0)
  INTO v_splits_sum
  FROM bank_splits
  WHERE tenant_id = NEW.tenant_id
   AND transaction_id = NEW.transaction_id
   AND (TG_OP <> 'UPDATE' OR id <> NEW.id);

  -- Adiciona o valor atual explicitamente
  v_splits_sum := v_splits_sum + NEW.amount_cents;

  IF v_splits_sum > v_transaction_amount THEN
    RAISE EXCEPTION 'SPLIT_AMOUNT_EXCEEDED: Soma dos splits (%) excede valor da transação (%)',
      v_splits_sum, v_transaction_amount
      USING ERRCODE = 'P0001';
  END IF;

  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_validate_split_consistency
  BEFORE INSERT OR UPDATE ON bank_splits
  FOR EACH ROW EXECUTE FUNCTION validate_split_consistency();

-- ============================================================
-- ATL CHECK
-- ============================================================

CREATE OR REPLACE FUNCTION check_atl_before_transaction()
RETURNS TRIGGER
SET search_path = pg_catalog, pg_temp
AS $$
BEGIN
  IF EXISTS (
    SELECT 1 FROM atl_blocked_actors 
    WHERE actor_id = NEW.actor_id
      AND tenant_id = NEW.tenant_id  -- HARD MODE: Filtro defensivo explícito
  ) THEN
    RAISE EXCEPTION 'ATL_BLOCKED' USING ERRCODE = 'P0001';
  END IF;
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

CREATE TRIGGER trg_check_atl
  BEFORE INSERT ON bank_transactions
  FOR EACH ROW EXECUTE FUNCTION check_atl_before_transaction();

-- ============================================================
-- PURPOSE VALIDATION
-- ============================================================

CREATE OR REPLACE FUNCTION validate_transfer_purpose()
RETURNS TRIGGER
SET search_path = pg_catalog, pg_temp
AS $$
BEGIN
  IF NEW.purpose IS NULL THEN
    RAISE EXCEPTION 'MISSING_PURPOSE' USING ERRCODE = 'P0001';

     AND (NEW.justification IS NULL OR LENGTH(TRIM(NEW.justification)) < 10) THEN
    RAISE EXCEPTION 'MISSING_JUSTIFICATION' USING ERRCODE = 'P0001';
  END IF;
  
  RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- ============================================================
-- EXECUTION FUND MOVEMENTS
-- ============================================================
--
-- DECISÃO ARQUITETURAL: Movimentos desacoplados de transactions
--
-- execution_fund_movements registra movimentações do fundo de execução
-- que podem ou não estar vinculadas a uma bank_transaction específica.
--
-- Design consciente:
--   - source_account_id é opcional (pode ser movimento sem origem específica)
--   - Não há transaction_id (movimento pode ser administrativo, não transacional)
--   - Permite registro de movimentações como:
--     * Expiração de créditos (não gera transaction)
--     * Recuperação de órfãos (não gera transaction)
--     * Execução administrativa (pode gerar transaction depois)
--
-- Se vinculação obrigatória a transaction for necessária no futuro:
--   - Adicionar transaction_id UUID
--   - Adicionar FK composta para bank_transactions
--
-- Por ora, mantemos desacoplado por design.
--

-- HARD MODE: FK composta cross-tenant (opcional, então pode ser NULL)
CREATE TABLE execution_fund_movements (
  id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
  tenant_id UUID NOT NULL REFERENCES tenants(id),
  movement_type TEXT NOT NULL CHECK (
    movement_type IN ('credit_expiration','orphan_recovery','execution','stabilization')
  ),
  amount_cents BIGINT NOT NULL CHECK (amount_cents > 0),
  source_account_id UUID, -- Removido REFERENCES direto
  source_description TEXT,
  approved_by_user_id UUID,
  approved_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  -- FK composta: proteção estrutural cross-tenant (apenas se source_account_id não for NULL)
  CONSTRAINT fk_execution_fund_movements_account 
    FOREIGN KEY (tenant_id, source_account_id) 
    REFERENCES bank_accounts(tenant_id, id),
  -- HARD MODE++: Rastreabilidade obrigatória baseada em tipo de movimento
  -- Garante que movimentos sem source_account_id tenham rastreabilidade alternativa
  CONSTRAINT execution_fund_movements_traceability
    CHECK (
      -- Caso 1: Tem source_account_id (rastreado via conta)
      (source_account_id IS NOT NULL)
      -- Caso 2: Não tem source_account, mas é tipo administrativo com descrição obrigatória
      OR (
        source_account_id IS NULL
        AND movement_type IN ('credit_expiration', 'orphan_recovery')
        AND source_description IS NOT NULL
        AND LENGTH(TRIM(source_description)) >= 20
      )
      -- Caso 3: Execução ou estabilização requer aprovação explícita
      OR (
        movement_type IN ('execution', 'stabilization')
        AND approved_by_user_id IS NOT NULL
        AND approved_at IS NOT NULL
      )
    )
);

CREATE INDEX idx_execution_fund_movements_tenant 
  ON execution_fund_movements(tenant_id);
-- HARD MODE: Índice composto para performance da FK composta (account)
-- Cobre todos os casos, incluindo NULLs (útil para queries gerais)
CREATE INDEX idx_execution_fund_movements_tenant_account
  ON execution_fund_movements(tenant_id, source_account_id);
-- HARD MODE+: Índice parcial para evitar bloat quando source_account_id é NULL
-- Otimizado para queries que filtram apenas registros com source_account_id não-nulo
-- Redundância controlada: ambos os índices servem propósitos diferentes
CREATE INDEX idx_execution_fund_movements_tenant_account_not_null
  ON execution_fund_movements(tenant_id, source_account_id)
  WHERE source_account_id IS NOT NULL;

-- AUDITORIA DE COBERTURA

-- HARD MODE++: Auditoria real de todas as verificações de cobertura aprovadas.

  UPDATE bank_accounts

  RETURN v_count;

  GET DIAGNOSTICS v_count = ROW_COUNT;

-- Fragmentation Detection Function
CREATE OR REPLACE FUNCTION detect_fragmentation(
  p_actor_id UUID,
  p_window_hours INTEGER DEFAULT 24,
  p_threshold_count INTEGER DEFAULT 10,
  p_small_amount_threshold_cents INTEGER DEFAULT 10000
) RETURNS TABLE(is_suspicious BOOLEAN, transaction_count INTEGER, total_amount_cents BIGINT, evidence JSONB)
SET search_path = pg_catalog, pg_temp
AS $$
DECLARE
  v_count INTEGER;
  v_total BIGINT;
  v_tenant_id UUID;
BEGIN
  SELECT tenant_id INTO v_tenant_id FROM actors WHERE id = p_actor_id;
  
  SELECT COUNT(*), COALESCE(SUM(amount_cents), 0) INTO v_count, v_total
  FROM bank_transactions bt
  JOIN bank_accounts ba ON ba.id = bt.account_id
  WHERE ba.actor_id = p_actor_id
    AND bt.created_at > now() - (p_window_hours || ' hours')::interval
    AND bt.amount_cents < p_small_amount_threshold_cents;
  
  IF v_count >= p_threshold_count THEN
    INSERT INTO evasion_patterns (tenant_id, actor_id, pattern_type, evidence, confidence_score)
    VALUES (v_tenant_id, p_actor_id, 'fragmentation',
      jsonb_build_object('count', v_count, 'total_cents', v_total),
      LEAST((v_count::NUMERIC / (p_threshold_count * 2) * 10000)::INTEGER, 10000));
    
    RETURN QUERY SELECT true, v_count, v_total, jsonb_build_object('pattern', 'fragmentation', 'count', v_count);
  ELSE
    RETURN QUERY SELECT false, v_count, v_total, NULL::JSONB;
  END IF;
END;
$$ LANGUAGE plpgsql;

-- Lei 2: Regime Forward-Only
-- Proíbe downgrade, reexecução e execução fora de ordem.

-- HARD MODE: Modo Constitucional Rígido (sem IF NOT EXISTS)
-- Contrato: Banco sempre provisionado do zero
-- ================================================

-- 1. Criação da autoridade única de versão
-- HARD MODE: Removido IF NOT EXISTS - modo constitucional rígido
-- Se tabela existir, migration deve falhar explicitamente

CREATE TABLE schema_version (
    version INTEGER PRIMARY KEY,
    applied_at TIMESTAMP NOT NULL DEFAULT CURRENT_TIMESTAMP,
    CONSTRAINT schema_version_positive CHECK (version > 0)
);

-- 2. Inserção da versão atual (Genesis + 0006)

INSERT INTO schema_version (version)
VALUES (6);

-- 3. Função de proteção estrutural

CREATE OR REPLACE FUNCTION enforce_forward_only()
RETURNS TRIGGER
SET search_path = pg_catalog, pg_temp
AS $$
DECLARE
    current_max INTEGER;
BEGIN
    SELECT MAX(version) INTO current_max FROM schema_version;

    -- Bloqueia reexecução ou versão fora da sequência
    IF TG_OP = 'INSERT' THEN
        IF NEW.version <> current_max + 1 THEN
            RAISE EXCEPTION
            'Forward-only violation: expected version %, got %',
            current_max + 1,
            NEW.version;
        END IF;
    END IF;

    -- Bloqueia qualquer UPDATE
    IF TG_OP = 'UPDATE' THEN
        RAISE EXCEPTION
        'Forward-only violation: UPDATE not allowed on schema_version';
    END IF;

    -- Bloqueia qualquer DELETE
    IF TG_OP = 'DELETE' THEN
        RAISE EXCEPTION
        'Forward-only violation: DELETE not allowed on schema_version';
    END IF;

    RETURN NEW;
END;
$$ LANGUAGE plpgsql;

-- 4. Trigger de proteção

CREATE TRIGGER schema_version_forward_only
BEFORE INSERT OR UPDATE OR DELETE
ON schema_version
FOR EACH ROW
EXECUTE FUNCTION enforce_forward_only();

-- Garantir KYC canônico
UPDATE identities
SET kyc_status = 'approved'
WHERE kyc_status = 'verified';

-- Remover valores não canônicos
UPDATE identities
SET kyc_status = 'pending'
WHERE kyc_status NOT IN ('pending','approved','rejected');
